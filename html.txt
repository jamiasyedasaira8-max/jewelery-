<script>
  // Product Data
  const products = [

    { id: 7, name: "Gold Tennis Bracelet", category: "bracelets", price: 2199, img: "⚜️", desc: "14K gold with 3ct total diamonds" },
    { id: 8, name: "Charm Bangle", category: "bracelets", price: 699, img: "⚜️", desc: "Sterling silver with 5 charms" },
    { id: 9, name: "Vintage Engagement Ring", category: "rings", price: 3299, img: "💍", desc: "Art deco style, 1.5ct center stone" },
    { id: 10, name: "Heart Locket Necklace", category: "necklaces", price: 499, img: "📿", desc: "Gold-plated with photo insert" },
    { id: 11, name: "Hoop Earrings", category: "earrings", price: 399, img: "✨", desc: "18K gold, medium size" },
    { id: 12, name: "Chain Link Bracelet", category: "bracelets", price: 899, img: "⚜️", desc: "Italian silver, adjustable" }
  ];
  
  let cart = JSON.parse(localStorage.getItem('luxeCart')) || [];
  
  // Navigation
  function showPage(pageId) {
    document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
    document.getElementById(pageId).classList.add('active');
    if (pageId === 'cart') renderCart();
    window.scrollTo(0, 0);
  }
  
  // Render Products
  function renderProducts(filter = 'all') {
    const grid = document.getElementById('productsGrid');
    const filtered = filter === 'all' ? products : products.filter(p => p.category === filter);
    
    grid.innerHTML = filtered.map(p => `
      <div class="product-card">
        <span class="product-img">${p.img}</span>
        <h3 class="product-name">${p.name}</h3>
        <p class="product-desc">${p.desc}</p>
        <p class="product-price">$${p.price.toLocaleString()}</p>
        <button class="add-btn" onclick="addToCart(${p.id})">ADD TO CART</button>
      </div>
    `).join('');
  }
  
  function filterProducts(category, btn) {
    document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    renderProducts(category);
  }
  
  // Cart Functions
  function addToCart(id) {
    const product = products.find(p => p.id === id);
    const existing = cart.find(item => item.id === id);
    
    if (existing) {
      existing.qty++;
    } else {
      cart.push({ ...product, qty: 1 });
    }
    
    saveCart();
    updateCartCount();
    
    // Visual feedback
    const btn = event.target;
    const originalText = btn.textContent;
    btn.textContent = '✓ ADDED';
    btn.style.background = '#4caf50';
    setTimeout(() => {
      btn.textContent = originalText;
      btn.style.background = '#d4af37';
    }, 1000);
  }
  
  function updateQty(id, change) {
    const item = cart.find(i => i.id === id);
    if (item) {
      item.qty += change;
      if (item.qty <= 0) {
        cart = cart.filter(i => i.id !== id);
      }
      saveCart();
      renderCart();
      updateCartCount();
    }
  }
  
  function removeFromCart(id) {
    cart = cart.filter(i => i.id !== id);
    saveCart();
    renderCart();
    updateCartCount();
  }
  
  function renderCart() {
    const container = document.getElementById('cartContainer');
    
    if (cart.length === 0) {
      container.innerHTML = `
        <div class="empty-cart">
          <p style="font-size:60px; margin-bottom:20px;">🛍️</p>
          <p>Your cart is empty.</p>
          <button class="btn" style="margin-top:20px;" onclick="showPage('shop')">START SHOPPING</button>
        </div>
      `;
      return;
    }
    
    const total = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
    const tax = total * 0.08;
    const grandTotal = total + tax;
    
    container.innerHTML = `
      ${cart.map(item => `
        <div class="cart-item">
          <span class="cart-item-img">${item.img}</span>
          <div>
            <div class="cart-item-name">${item.name}</div>
            <div style="color:#c9b896; font-size:13px; font-style:italic;">${item.desc}</div>
          </div>
          <div class="cart-item-price">$${item.price.toLocaleString()}</div>
          <div class="qty-controls">
            <button class="qty-btn" onclick="updateQty(${item.id}, -1)">−</button>
            <span style="min-width:25px; text-align:center;">${item.qty}</span>
            <button class="qty-btn" onclick="updateQty(${item.id}, 1)">+</button>
          </div>
          <button class="remove-btn" onclick="removeFromCart(${item.id})" title="Remove">✕</button>
        </div>
      `).join('')}
      
      <div class="cart-summary">
        <p style="margin-bottom:10px;">Subtotal: <span style="color:#f5e6d3;">$${total.toLocaleString()}</span></p>
        <p style="margin-bottom:10px;">Tax (8%): <span style="color:#f5e6d3;">$${tax.toFixed(2)}</span></p>
        <h3>Total: $${grandTotal.toFixed(2)}</h3>
        <button class="btn" onclick="checkout()" style="margin-top:15px;">PROCEED TO CHECKOUT</button>
      </div>
    `;
  }
  
  function checkout() {
    if (cart.length === 0) return;
    alert('🎉 Thank you for your purchase!\n\nYour order has been placed successfully.\nA confirmation email will be sent shortly.');
    cart = [];
    saveCart();
    updateCartCount();
    renderCart();
  }
  
  function saveCart() {
    localStorage.setItem('luxeCart', JSON.stringify(cart));
  }
  
  function updateCartCount() {
    const count = cart.reduce((sum, item) => sum + item.qty, 0);
    document.getElementById('cartCount').textContent = count;
  }
  
  // Contact Form
  function submitForm(e) {
    e.preventDefault();
    const msg = document.getElementById('successMsg');
    msg.classList.add('show');
    document.getElementById('contactForm').reset();
    setTimeout(() => msg.classList.remove('show'), 5000);
  }
  
  // Initialize
  renderProducts();
  updateCartCount();
</script>
